    ```csharp
    /// <summary>CPT-03 — une seule instance de l'overlay par session Windows. POURQUOI un mutex nommé et pas un fichier-verrou : l'OS libère le mutex à la
    /// mort du processus (un fichier survivrait à un crash et bloquerait le redémarrage) ; POURQUOI Local\ : pas de droits, un autre utilisateur a le sien.
    /// Posé APRÈS les court-circuits --statusline / --hook / --cadrans / --sessions et AVANT le Host : les hooks (jusqu'à 5 en parallèle) restent multi-instances.</summary>
    public static class VerrouInstanceUnique
    {
        public const string NomOverlay = @"Local\Chronos-overlay";
        private static readonly ConcurrentDictionary<string, byte> Tenus = new(StringComparer.Ordinal);   // noms tenus par CE processus (diagnostic)

        public static ResultatVerrou Acquerir(string nom)
        {
            var m = new Mutex(initiallyOwned: true, nom, out var cree);
            if (cree) { Tenus[nom] = 1; return new ResultatVerrou(nom, true, m, false); }
            try
            {
                if (m.WaitOne(0)) { Tenus[nom] = 1; return new ResultatVerrou(nom, true, m, false); }
                m.Dispose(); return new ResultatVerrou(nom, false, null, false);
            }
            catch (AbandonedMutexException) { Tenus[nom] = 1; return new ResultatVerrou(nom, true, m, abandonne: true); }
        }

        /// <summary>« tenu par ce processus » | « tenu par un autre processus » | « libre » — sans jamais acquérir (TryOpenExisting).</summary>
        public static string EtatPourDiagnostic(string nom)
        {
            if (Tenus.ContainsKey(nom)) return "tenu par ce processus";
            return Mutex.TryOpenExisting(nom, out var m) ? Dispose(m, "tenu par un autre processus") : "libre";
        }
        internal static void Oublier(string nom) => Tenus.TryRemove(nom, out _);
    }
    ```
